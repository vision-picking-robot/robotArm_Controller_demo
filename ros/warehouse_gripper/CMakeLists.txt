cmake_minimum_required(VERSION 3.16)
project(warehouse_gripper LANGUAGES CXX)

find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(dsr_common2 REQUIRED)
find_package(dsr_hardware2 REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(Threads REQUIRED)

add_library(${PROJECT_NAME} SHARED src/gripper_controller.cpp)
target_compile_features(${PROJECT_NAME} PRIVATE cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME}
  controller_interface dsr_hardware2 pluginlib rclcpp std_srvs)
target_link_libraries(${PROJECT_NAME} dsr_common2::drfl Threads::Threads)
pluginlib_export_plugin_description_file(controller_interface gripper_plugin.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(gripper_protocol_test test/test_protocol.cpp)
  target_compile_features(gripper_protocol_test PRIVATE cxx_std_17)
  target_include_directories(gripper_protocol_test PRIVATE include)
  add_test(NAME gripper_protocol_test COMMAND gripper_protocol_test)
  add_executable(gripper_rtu_test test/test_rtu_session.cpp)
  target_compile_features(gripper_rtu_test PRIVATE cxx_std_17)
  target_include_directories(gripper_rtu_test PRIVATE include)
  target_link_libraries(gripper_rtu_test Threads::Threads)
  add_test(NAME gripper_rtu_test COMMAND gripper_rtu_test)
endif()
ament_package()
